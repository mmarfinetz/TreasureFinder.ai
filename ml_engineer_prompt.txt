from typing import Dict
import numpy as np

def calculate_indices(red: np.ndarray, green: np.ndarray, nir: np.ndarray, swir1: np.ndarray, blue: np.ndarray) -> Dict[str, np.ndarray]:
    small_eps: float = 1e-6
    ndvi = (nir - red) / (nir + red + small_eps)
    ndwi = (green - nir) / (green + nir + small_eps)
    bsi = ((swir1 + red) - (nir + blue)) / ((swir1 + red) + (nir + blue) + small_eps)
    return {"ndvi": ndvi, "ndwi": ndwi, "bsi": bsi}
```

def train_with_pseudo_labels(model, labeled_data, unlabeled_data):
    # Initial training on labeled data
    model.fit(labeled_data.X, labeled_data.y)
    
    # Generate pseudo-labels for high-confidence predictions
    predictions = model.predict_proba(unlabeled_data.X)
    high_confidence_mask = (predictions.max(axis=1) > 0.9)
    pseudo_labels = predictions.argmax(axis=1)[high_confidence_mask]
    
    # Retrain with combined dataset
    combined_X = np.vstack([labeled_data.X, 
                           unlabeled_data.X[high_confidence_mask]])
    combined_y = np.hstack([labeled_data.y, pseudo_labels])
    model.fit(combined_X, combined_y)
```
